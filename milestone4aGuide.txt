src/lambda/queues/FollowFetcherLambda.ts
Responsibility:

event contains a single Post message: { authorAlias, status: { post, timestamp, ... } }

Query follow DAO for all followers of author (you likely already have getAllFollowers(authorAlias) in follow DAO).

Chunk follower list into jobs (size JOB_CHUNK_SIZE, e.g., 250). For each chunk, send message to JobQueue containing { authorAlias, statusDto, followerAliases: [...] }.

File sketch (TypeScript):

import { SQSEvent } from "aws-lambda";
import { DynamoDaoFactory } from "../../../dynamo_daos/DynamoDaoFactory";
import { SQSClient, SendMessageCommand } from "@aws-sdk/client-sqs";
import { JobQueueUrl } from "../../DynamoConstants"; // or env var

const JOB_CHUNK_SIZE = 250;

export const handler = async (event: SQSEvent) => {
  const factory = new DynamoDaoFactory();
  const followDao = factory.getFollowDao();
  const sqs = new SQSClient({});

  for (const record of event.Records) {
    const body = JSON.parse(record.body);
    const { authorAlias, status } = body;

    const followers = await followDao.getAllFollowers(authorAlias); // returns string[]
    for (let i = 0; i < followers.length; i += JOB_CHUNK_SIZE) {
      const chunk = followers.slice(i, i + JOB_CHUNK_SIZE);
      await sqs.send(new SendMessageCommand({
        QueueUrl: process.env.JOB_QUEUE_URL!,
        MessageBody: JSON.stringify({ authorAlias, status, followerAliases: chunk })
      }));
    }
  }
};

File sketch:

import { SQSEvent } from "aws-lambda";
import { DynamoDbClientWrapper } from "../../../dynamo_daos/DynamoDbClientWrapper";
import { FeedTable } from "../../../dynamo_daos/DynamoConstants";

const BATCH_WRITE_SIZE = 25;

export const handler = async (event: SQSEvent) => {
  const db = new DynamoDbClientWrapper();

  for (const record of event.Records) {
    const job = JSON.parse(record.body);
    const { status, followerAliases } = job;

    // create items per follower
    const items = followerAliases.map(alias => ({
      [FeedTable.PK]: alias,
      [FeedTable.SK]: status.timestamp,
      [FeedTable.ATTR_POST]: status.post,
      [FeedTable.ATTR_USER]: JSON.stringify(status.user),
      [FeedTable.ATTR_AUTHOR]: status.user.alias,
    }));

    await db.batchWrite(FeedTable.TABLE, items); // ensure wrapper handles 25 chunking and retries
  }
};
